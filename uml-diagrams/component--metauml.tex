\documentclass[border=4pt]{standalone}
\usepackage{luamplib}
% Typeset all MetaUML labels with TeX, i.e., in the font of the document
\mplibtextextlabel{enable}
\begin{document}
\sffamily
\begin{mplibcode}
input metauml;
beginfig(1);
pair zTwo, zThree;
Component.cOne("component 1")();
Component.cTwo("component 2")();
Component.cThree("component 3")();
% MetaUML has no automatic layout: each object is placed relative to another one.
% The left edge of component 2 (3) is 140 PostScript points (1/72 inch, about 5 cm) right of and 40 points above (below) the right edge of component 1.
cTwo.w = cOne.e + (140, 40);
cThree.w = cOne.e + (140, -40);
drawObjects(cOne, cTwo, cThree);
% The interfaces are halfway between the components.
zTwo = cOne.e + (70, 40);
zThree = cOne.e + (70, -40);
link(providedInterface)(cOne.e -- zTwo);
link(requiredInterface)(cTwo.w -- zTwo);
item(iAssoc)("Interface 2")(obj.s = zTwo + (0, 8));
link(providedInterface)(cOne.e -- zThree);
link(requiredInterface)(cThree.w -- zThree);
item(iAssoc)("Interface 3")(obj.n = zThree - (0, 8));
endfig;
\end{mplibcode}
\end{document}
